# To run:
* Update repository:
chmod +x git-update2.sh
./git-update2.sh ~/Projects

- or
./git-update.sh ~/Projects/myrepo

* Push update:
chmod +x git-push-all2.sh
./git-push-all2.sh ~/Projects

* Displays currently logged-in users in a readable format.
chmod +x who-connected.sh
./who-connected2.sh
